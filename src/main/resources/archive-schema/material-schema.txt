CREATE TABLE HGAPPSAM.THGSAM_ORDER_MATERIAL (
	ORDER_MATERIAL_ID NUMBER,
	ORDER_ID NUMBER,
	OBJECT_ID NUMBER,
	QUANTITY NUMBER,
	DESCRIPTION VARCHAR2(200),
	OBJECT_TYPE VARCHAR2(20) DEFAULT 'MARKET_MATERIAL',
	WEB_VISIBLE NUMBER(1,0) DEFAULT 1,
	PRODUCT_TYPE VARCHAR2(30),
	PROD_ID VARCHAR2(30),
	DISTRIBUTION_CHANNEL NUMBER,
	DISTRIBUTION_TYPE NUMBER,
	CONSTRAINT SYS_C0011421 CHECK ("ORDER_ID" IS NOT NULL),
	CONSTRAINT SYS_C0011422 CHECK ("OBJECT_ID" IS NOT NULL),
	CONSTRAINT THGSAM_ORDER_MATERIAL_C1 CHECK (web_visible in (1,0)),
	CONSTRAINT THGSAM_ORDER_MATERIAL_PK1 PRIMARY KEY (ORDER_MATERIAL_ID),
	CONSTRAINT THGSAM_ORDER_MATERIAL_FK1 FOREIGN KEY (ORDER_ID) REFERENCES HGAPPSAM.THGSAM_ORDER(ORDER_ID)
);
CREATE INDEX THGSAM_ORDER_MATERIAL_N1 ON HGAPPSAM.THGSAM_ORDER_MATERIAL (ORDER_ID);
CREATE UNIQUE INDEX THGSAM_ORDER_MATERIAL_PK1 ON HGAPPSAM.THGSAM_ORDER_MATERIAL (ORDER_MATERIAL_ID);
