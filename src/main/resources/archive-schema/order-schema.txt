CREATE TABLE HGAPPSAM.THGSAM_ORDER (
	ORDER_ID NUMBER,
	CREATED_ON DATE,
	CREATED_BY_ID NUMBER,
	APPROVED_ON DATE,
	APPROVED_BY_ID NUMBER,
	ORDER_STATUS VARCHAR2(15),
	SHIPPING_TYPE VARCHAR2(15),
	SHIPPING_APPROVED_BY_ID NUMBER,
	SHIPPING_APPROVED_ON DATE,
	COMMENTS VARCHAR2(2000),
	ORDER_TYPE VARCHAR2(50),
	SHIPPED_ON DATE,
	LOW_PLAN_COUNT NUMBER(1,0),
	APP_ID NUMBER,
	WEB_SERVICE_FLAG NUMBER(1,0) DEFAULT 0 ,
	EFULFILLMENT_FLAG NUMBER(1,0) DEFAULT 0,
	BLAST_ID NUMBER,
	COM_DIV_SHIPPING_METHOD_ID NUMBER,
	WANT_DATE DATE,
	BUSINESS_JUSTIFICATION VARCHAR2(1000),
	ORDERED_ON_BEHALF_BY_ID NUMBER,
	ORDER_REPORTED_FLAG VARCHAR2(1) DEFAULT 'N',
	CREATED_BY_TERRITORY_CODE VARCHAR2(240),
	CHARGEBACK_NUM VARCHAR2(255),
	EFULFILMENT_STATUS VARCHAR2(50),
	TOTAL_COST NUMBER(10,2),
	REPORT_SENT NUMBER(1,0),
	CC_CORRELATION_ID NUMBER,
	TOTAL_COST_CHARGED NUMBER(10,2),
	SF_BATCH_ID NUMBER,
	POLICY_ORDER NUMBER(1,0),
	AOC_ORDER NUMBER(1,0),
	UPDATED_BY NUMBER,
	UPDATED_ON DATE,
	UPDATE_REASON VARCHAR2(300),
	CFS_FLAG NUMBER(1,0),
	CFS_STATUS VARCHAR2(25),
	CREATED_BY_PROCESS VARCHAR2(40),
	SENDCONFIRMATIONEMAIL NUMBER(1,0),
	POA_WINDOW_ID NUMBER,
	CARRIER_ACCT_NUM VARCHAR2(255),
	UPDATED_BY_PROCESS VARCHAR2(250),
	PO_NUMER VARCHAR2(50),
	CONSTRAINT SYS_C0011425 CHECK ("ORDER_ID" IS NOT NULL),
	CONSTRAINT SYS_C0011426 CHECK ("CREATED_BY_ID" IS NOT NULL),
	CONSTRAINT SYS_C0011427 CHECK ("WEB_SERVICE_FLAG" IS NOT NULL),
	CONSTRAINT THGSAM_ORDER_PK PRIMARY KEY (ORDER_ID)
);
CREATE INDEX THGSAM_ORDER_B1 ON HGAPPSAM.THGSAM_ORDER (CREATED_BY_PROCESS);
CREATE INDEX THGSAM_ORDER_N1 ON HGAPPSAM.THGSAM_ORDER (ORDER_STATUS);
CREATE INDEX THGSAM_ORDER_N10 ON HGAPPSAM.THGSAM_ORDER (POA_WINDOW_ID);
CREATE INDEX THGSAM_ORDER_N2 ON HGAPPSAM.THGSAM_ORDER (APP_ID,ORDER_STATUS);
CREATE UNIQUE INDEX THGSAM_ORDER_PK ON HGAPPSAM.THGSAM_ORDER (ORDER_ID);
